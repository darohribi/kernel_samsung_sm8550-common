# SM8550 Maximum Performance Kernel Command Line
# Add to BOOT_IMAGE_CMDLINE or kernel command line in device tree / boot.img
# Each parameter explained with expected impact

# =========================================================================
# SCHEDULER / CPU
# =========================================================================
# Disable scheduler autogroup (reduces overhead, better for gaming)
sched_autogroup_enabled=0

# Disable scheduler debug stats (small overhead reduction)
# # sched_debug=0  # DISABLED: keep sched_debug for boot diagnostics

# Reduce migration cost for faster task placement
sched_migration_cost_ns=500000

# Increase wakeup granularity for throughput
sched_wakeup_granularity_ns=1000000

# Disable RT runtime limiting (allow full CPU for RT tasks)
# # sched_rt_runtime_us=-1  # DISABLED: unlimited RT breaks init boot priority

# =========================================================================
# MEMORY MANAGEMENT
# =========================================================================
# Transparent Hugepages: always (maximum throughput)
transparent_hugepage=madvise

# Page allocator: prefer local node
numa_balancing=0

# Disable memory compaction overhead (may increase fragmentation slightly)
# compact_unevictable_allowed=0

# Slab: disable debug, enable performance
# # slub_debug=-  # DISABLED: removes all poisoning, causes silent corruption
# # slub_max_order=3  # DISABLED: default 0, higher = boot-time OOM on phones
slub_min_objects=16

# =========================================================================
# RCU
# =========================================================================
# Accelerate RCU grace periods
# # rcu_nocbs=0-7  # DISABLED: all-CPU nocb breaks kthreadd boot
rcu_nocb_poll=1

# =========================================================================
# CPU IDLE / FREQUENCY
# =========================================================================
# Use TEO governor (better latency than menu for interactive)
cpuidle.off=0
cpuidle.governor=teo

# Disable intel_idle (not on ARM, but safe)
intel_idle.max_cstate=0

# =========================================================================
# BLOCK / I/O
# =========================================================================
# Disable I/O scheduler merges (lower latency)
scsi_mod.use_blk_mq=1

# =========================================================================
# DEBUG / TRACING OVERHEAD REDUCTION
# =========================================================================
# Disable ftrace function tracer
ftrace=function

# Disable printk timestamp overhead
printk.time=0

# Reduce console log level
loglevel=3

# Disable lockdep (already off in defconfig, but ensure)
lockdep=0

# Disable debug objects
debug_objects=0

# Disable kmemleak
kmemleak=off

# =========================================================================
# THERMAL / POWER
# =========================================================================
# Disable thermal throttling early (use userspace governor instead)
thermal.off=0
thermal.nocrt=1

# =========================================================================
# KERNEL PRE-EMPTION
# =========================================================================
# Already PREEMPT=y in config; these are runtime
preempt=full

# =========================================================================
# ANDROID SPECIFIC
# =========================================================================
# Disable Android debug overhead
androidboot.debug=0
androidboot.verifiedbootstate=green

# =========================================================================
# COMBINED SINGLE LINE (for boot.img cmdline)
# =========================================================================
